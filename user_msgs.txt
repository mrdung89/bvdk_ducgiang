    def load_sessions(self):
        dt = self.view.dt_date.date().toString("yyyy-MM-dd")
        self.view.cb_sessions.clear()
        self.current_sessions.clear()
        
        try:
            # 1. Bổ sung select cột ma_phieu 
            # 2. Lấy cả trạng thái DA_TIEP_NHAN và DANG_GIAT (đã được KSNK duyệt)
            rows = self.db.fetch_all("""
                SELECT id, khoa_giao, ma_do, so_luong, DATE_FORMAT(thoi_gian, '%H:%i') as tm, ma_phieu 
                FROM lich_su_giao_nhan 
                WHERE DATE(thoi_gian) = %s AND trang_thai IN ('DA_TIEP_NHAN', 'DANG_GIAT')
            """, (dt,))
            
            # Group by Mã Phiên (nếu có) hoặc fallback về Khoa - Thời gian
            for r in rows:
                ma_phien = r.get('ma_phieu')
                if ma_phien:
                    key = f"Mã phiên: {ma_phien} ({r['khoa_giao']} - {r['tm']})"
                else:
                    key = f"{r['khoa_giao']} - {r['tm']}"
                    
                if key not in self.current_sessions:
                    self.current_sessions[key] = []
                self.current_sessions[key].append(r)
                
            for key in self.current_sessions.keys():
                self.view.cb_sessions.addItem(key)
                
        except Exception as e:
            print("Error loading sessions:", e)
